Algorithm: Super-Droplet Backward Tracking Algorithm

// Initialization phase
Procedure INITIALIZE_SD_TRACKING()
    FOR each super-droplet i DO
        pre_sdid[i] ← current_sdid[i]     // Record previous SD identifier
        pre_dmid[i] ← current_dmid[i]     // Record previous domain identifier
        if_coal[i] ← 0                    // Initialize coalescence flag
    END FOR
END PROCEDURE

// Main tracking loop
Procedure SD_BACKWARD_TRACKING(time_step)
    // 1. State recording and updating
    FOR each super-droplet i DO
        // Save current state as "previous state" for next time step
        temp_sdid ← current_sdid[i]
        temp_dmid ← current_dmid[i]
        
        // Update tracking information
        pre_sdid[i] ← temp_sdid
        pre_dmid[i] ← temp_dmid
        if_coal[i] ← 0  // Reset coalescence flag
    END FOR
    
    // 2. Motion tracking (simplified)
    CALL SD_MOTION_TRACKING()
    
    // 3. Coalescence process tracking
    CALL SD_COALESCENCE_TRACKING()
    
    // 4. Boundary condition handling
    CALL SD_BOUNDARY_HANDLING()
    
    // 5. Output tracking data
    IF (output_time) THEN
        CALL OUTPUT_SD_STATE_DATA()  // Output at specified intervals
    END IF
END PROCEDURE

// Motion tracking subroutine (simplified)
Procedure SD_MOTION_TRACKING()
    FOR each super-droplet i DO
        // Update position using velocity interpolation and terminal velocity
        UPDATE_SD_POSITION(i)
    END FOR
END PROCEDURE

// Coalescence event tracking subroutine
Procedure SD_COALESCENCE_TRACKING()
    // Sort super-droplets by grid
    CALL SORT_SD_BY_GRID()
    
    FOR each grid cell DO
        sd_list ← GET_SD_IN_CELL()
        
        // Random pairing
        pairs ← RANDOM_PAIRING(sd_list)
        
        FOR each pair (i, j) in pairs DO
            // Calculate coalescence probability
            prob_collision ← CALCULATE_COLLISION_PROBABILITY(
                sd_r[i], sd_r[j], sd_n[i], sd_n[j], 
                relative_velocity, collection_efficiency)
            
            // Random number test for coalescence occurrence
            IF (RANDOM() < prob_collision) THEN
                // Record detailed coalescence event information before coalescence
                IF (sd_n[i] > sd_n[j]) THEN
                    // Record SD with larger multiplicity as first participant
                    pre_sdid1_temp ← pre_sdid[i]  // Previous ID of SD with larger multiplicity
                    pre_dmid1_temp ← pre_dmid[i]  // Previous domain ID SD with larger multiplicity
                    pre_sdid2_temp ← pre_sdid[j]  // Previous ID of SD with smaller multiplicity
                    pre_dmid2_temp ← pre_dmid[j]  // Previous domain ID SD with smaller multiplicity
                    sd_n1 ← sd_n[i]
                    sd_r1 ← sd_r[i]
                    sd_n2 ← sd_n[j]
                    sd_r2 ← sd_r[j]
                ELSE
                    // Record SD with larger multiplicity as first participant
                    pre_sdid1_temp ← pre_sdid[j]  // Previous ID of SD with larger multiplicity
                    pre_dmid1_temp ← pre_dmid[j]  // Previous domain ID SD with larger multiplicity
                    pre_sdid2_temp ← pre_sdid[i]  // Previous ID of SD with smaller multiplicity
                    pre_dmid2_temp ← pre_dmid[i]  // Previous domain ID SD with smaller multiplicity
                    sd_n1 ← sd_n[j]
                    sd_r1 ← sd_r[j]
                    sd_n2 ← sd_n[i]
                    sd_r2 ← sd_r[i]
                END IF

                // Calculate mass (volume) for coalescence computation
                sd_m1 ← sd_r1^3  // Volume proportional to mass
                sd_m2 ← sd_r2^3  // Volume proportional to mass

                // Handle coalescence based on multiplicity comparison
                IF (sd_n1 > sd_n2 * sd_ncol) THEN
                    // Different multiplicities: larger one partially coalesces
                    sd_n1 ← sd_n1 - sd_n2 * sd_ncol
                    sd_r2 ← (sd_m1 * sd_ncol + sd_m2)^(1/3)
                ELSE
                    // Same multiplicities: split equally to avoid complete disappearance
                    sd_n1 ← sd_n2 / 2
                    sd_n2 ← sd_n2 - sd_n1
                    sd_r1 ← (sd_m1 * sd_ncol + sd_m2)^(1/3)  // Same radius
                    sd_r2 ← sd_r1
                END IF
                
                num_col_temp ← CALCULATE_COLLISION_NUMBER(sd_n[i], sd_n[j])
                
                // Perform coalescence operation
                PERFORM_COALESCENCE(i, j)
                
                // Important: pre_sdid and pre_dmid values are NOT modified
                // They remain unchanged to preserve tracking continuity
                
                // Set coalescence flags
                if_coal[i] ← 1
                if_coal[j] ← 1
                
                // Store coalescence event data and output immediately
                STORE_AND_OUTPUT_COALESCENCE_EVENT(
                    pre_sdid1_temp, pre_sdid2_temp,
                    pre_dmid1_temp, pre_dmid2_temp,
                    num_col_temp, result_radius, result_multiplicity)
            END IF
        END FOR
    END FOR
END PROCEDURE

// Boundary condition handling
Procedure SD_BOUNDARY_HANDLING()
    FOR each super-droplet i DO
        // Check boundary conditions
        IF (sd_x[i] < x_min OR sd_x[i] > x_max OR
            sd_y[i] < y_min OR sd_y[i] > y_max OR
            sd_z[i] < z_min OR sd_z[i] > z_max) THEN
            
            // Handle boundary crossing
            HANDLE_BOUNDARY_CROSSING(i)
            
            // MPI communication for tracking information
            IF (crosses_domain_boundary) THEN
                SEND_TRACKING_DATA(pre_sdid[i], pre_dmid[i], if_coal[i])
            END IF
        END IF
    END FOR
END PROCEDURE

// Output SD state data (at specified output intervals)
Procedure OUTPUT_SD_STATE_DATA()
    // Output super-droplet state variables:
    // - Spatial coordinates: sd_x, sd_y, sd_z
    // - Physical properties: sd_r (radius), sd_n (multiplicity)
    // - Velocity: sd_vz (terminal velocity)
    // - Aerosol properties: sd_asl (aerosol mass)
    // - Phase status: sd_liqice (liquid/ice status)
    // - Tracking variables: pre_sdid, pre_dmid, if_coal
    WRITE_TO_OUTPUT_FILE(
        sd_x, sd_y, sd_z, sd_r, sd_n, sd_vz, sd_asl, sd_liqice,
        pre_sdid, pre_dmid, if_coal)
END PROCEDURE

// Store and output coalescence event data (at every coalescence time step)
Procedure STORE_AND_OUTPUT_COALESCENCE_EVENT(
    pre_sdid1, pre_sdid2, pre_dmid1, pre_dmid2, num_col, result_radius, result_multiplicity)
    
    // Output coalescence event variables:
    // - pre_sdid1: Previous SD ID of first colliding droplet
    // - pre_sdid2: Previous SD ID of second colliding droplet  
    // - pre_dmid1: Previous domain ID of first colliding droplet
    // - pre_dmid2: Previous domain ID of second colliding droplet
    // - num_col: Number of coalescence events
    // - result_radius: Radius of resulting droplet after coalescence
    // - result_multiplicity: Multiplicity of resulting droplet
    
    WRITE_COALESCENCE_TO_NETCDF(
        pre_sdid1, pre_sdid2, pre_dmid1, pre_dmid2, 
        num_col, result_radius, result_multiplicity)
END PROCEDURE

// Main program entry
Procedure MAIN()
    CALL INITIALIZE_SD_TRACKING()
    
    FOR time_step = 1 TO max_time_steps DO
        CALL SD_BACKWARD_TRACKING(time_step)
    END FOR
END PROCEDURE
